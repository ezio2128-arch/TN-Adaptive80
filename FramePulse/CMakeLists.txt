cmake_minimum_required(VERSION 3.24)
project(FramePulse VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
add_executable(FramePulseTests Tests/engine_tests.cpp)
target_include_directories(FramePulseTests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
add_test(NAME adaptive COMMAND FramePulseTests)
if(WIN32)
  enable_language(RC)
  if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Windows requires MSVC x64 and Windows SDK C++/WinRT")
  endif()
  add_library(MinHook STATIC ThirdParty/minhook/src/buffer.c ThirdParty/minhook/src/hook.c
    ThirdParty/minhook/src/trampoline.c ThirdParty/minhook/src/hde/hde64.c)
  target_include_directories(MinHook PUBLIC ThirdParty/minhook/include)
  add_library(FramePulsePacing SHARED Pacing/Hook.cpp)
  target_link_libraries(FramePulsePacing PRIVATE MinHook d3d11 dxgi advapi32)
  add_executable(FramePulseCore WIN32 Core/Main.cpp Core/FramePulse.rc)
  target_link_libraries(FramePulseCore PRIVATE windowsapp user32 shell32 advapi32 dxgi)
  add_executable(FramePulseAttach Pacing/Attach.cpp)
  target_link_libraries(FramePulseAttach PRIVATE advapi32)
  add_executable(FramePulseTestGame WIN32 Tests/TestGame.cpp)
  target_link_libraries(FramePulseTestGame PRIVATE d3d11 dxgi)
  foreach(t FramePulseCore FramePulsePacing FramePulseAttach FramePulseTestGame)
    target_include_directories(${t} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/Assets)
    target_compile_definitions(${t} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_compile_options(${t} PRIVATE /W4 /EHsc)
    set_property(TARGET ${t} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endforeach()
endif()
